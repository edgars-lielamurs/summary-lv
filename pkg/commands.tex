% Store supervisor metadata in arrays: '\@SupervisorNames', '\@SupervisorDegrees', '\@SupervisorPositions'
\newcounter{SupervisorCounter}
\newcommand{\addsupervisor}[3]{%
  \stepcounter{SupervisorCounter}%
  \csdef{SupervisorNames\csuse{theSupervisorCounter}}{#1}%
  \csdef{SupervisorDegrees\csuse{theSupervisorCounter}}{#2}%
  \csdef{SupervisorPositions\csuse{theSupervisorCounter}}{#3}%
}

\newcommand\getSupervisorName[1]{
  \csuse{SupervisorNames#1}%
}
\newcommand\getSupervisorDegree[1]{
  \csuse{SupervisorDegrees#1}%
 }
\newcommand\getSupervisorPosition[1]{
  \csuse{SupervisorPositions#1}%
 }

% nonenumerated section

  \addcontentsline{toc}{section}{#1}
}
